def _subir(self, i):
    ids, chaves, pos = self.ids, self.chaves, self.pos
    ident, chave = ids[i], chaves[i]
    while i > 0:
        pai = (i - 1) >> 1
        if chave < chaves[pai]:
            ids[i] = ids[pai]
            chaves[i] = chaves[pai]
            pos[ids[i]] = i
            i = pai
        else:
            break
    ids[i] = ident
    chaves[i] = chave
    pos[ident] = i

def alterar_chave(self, ident, nova):
    i = self.pos[ident]
    antiga = self.chaves[i]
    self.chaves[i] = nova
    if nova < antiga:
        self._subir(i)
    elif nova > antiga:
        self._descer(i)
